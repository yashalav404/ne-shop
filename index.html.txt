        const tg = window.Telegram.WebApp;
        tg.expand(); // Разворачивает приложение на весь экран

        function checkId() {
            const playerImg = document.getElementById('pubg-id').value;
            if(!playerImg) {
                alert("Пожалуйста, введите ID");
                return;
            }

            // Имитируем проверку на сайте. Когда подключим API, тут будет реальный запрос
            document.getElementById('nickname').innerText = "Аккаунт найден: 😎 СУПЕР_ПРО_ИГРОК";
            document.getElementById('nickname').style.display = "block";
            
            // Показываем кнопку оплаты СБП
            document.getElementById('sbp-btn').style.display = "block";
        }

        function payWithSBP() {
            const playerId = document.getElementById('pubg-id').value;
            
            // Отправляем данные обратно в нашего Python-бота
            const data = {
                id: playerId,
                item: "60 UC",
                price: 65
            };
            tg.sendData(JSON.stringify(data)); // Бот получит эту строку
            tg.close(); // Закрываем мини-приложение
        }
